## C++
```cpp
#include<iostream>
#include<vector>
#include<algorithm>
#include<cmath>
#include<set>

using namespace std;

#define pdd pair<double, double>

int grace_cover(vector<pdd> &segs, double l) 
{
    sort(segs.begin(), segs.end());
    double current_end = 0, farthest_end = 0;
    int num_sprinklers = 0, i = 0;
    int n = segs.size();
    
    while (current_end < l) 
    {
        while (i < n && segs[i].first <= current_end) 
        {
            farthest_end = max(farthest_end, segs[i].second);
            i++;
        }
        if (current_end == farthest_end) 
            return -1;
        current_end = farthest_end;
        num_sprinklers++;
    }
    
    return num_sprinklers;
}

int main() 
{
    int n, l;
    uint32_t w, p, r;
    
    while (cin >> n >> l >> w) 
    {
        set<pdd> unique_segs;
        for (int i = 0; i < n; i++) 
        {
            cin >> p >> r;
            if (2 * r <= w) 
                continue;
            double half_len = sqrt(r * r - (w * w / 4.0));
            unique_segs.emplace(p - half_len, p + half_len);
        }
        vector<pdd> segs(unique_segs.begin(), unique_segs.end());
        cout << grace_cover(segs, l) << endl;
    }
    return 0;
}
```
