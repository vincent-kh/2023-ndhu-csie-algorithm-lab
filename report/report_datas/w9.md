    ```cpp
    vector<pair<int,int>> wv;
    vector<int> dp(mx+1,-1);
    dp[0]=0;
    for(int i=0;i<wv.size();i++)    
    {
        for(int j=wv[i].first;j<=mx;j++)
        {
            if(dp[j-wv[i].first]!=-1 && dp[j-wv[i].first]+wv[i].second>dp[j])
            {
                dp[j]=dp[j-wv[i].first]+wv[i].second
            }
        }
    }
    ```

3. Design an algorithm to reconstruct the combination of items with $O(n \times W)$ table, and analyzing its time complexity.     
    ```cpp
    int dp[wv.size()+1][mx+1];
    for(int i=0;i<wv.size()+1;i++)
        dp[i][0]=0;
    for(int j=0;j<mx+1;j++)
        dp[0][j]=0;
    for(int i=1;i<wv.size()+1;i++)
        for(int j=w[i];j<mx+1;j++)
            dp[i][j] = max(dp[i-1][j] , dp[i][j-w[i]]+v[i])
    ```

## C++
```cpp
#include<iostream>
#include<vector>
using namespace std;
int main()
{
    int burger[2], t;

    while (cin>>burger[0]>>burger[1]>>t)
    {
        int i, j;
        vector<int> eat(t+1,-1);
        eat[0] = 0;

        for (i = 0; i < 2; i++)
        {
            for (j = burger[i]; j <= t; j++)
            {
                
                if (eat[j-burger[i]] != -1 && eat[j-burger[i]] + 1 > eat[j])
                {    
                    eat[j] = eat[j - burger[i]] + 1;
                }
            }
        }
        for (i = t; i >= 0; i--)
        {
            if (eat[i] != -1)
            {
                cout<<eat[i]<<' ';
                break;
            }
        }

        if (i != t)
            cout<<t - i;
        cout<<endl;
    }
    return 0;
}
```
